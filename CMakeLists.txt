cmake_minimum_required(VERSION 3.22)

project(JPClipper VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_subdirectory(JUCE)

if (UNIX AND NOT APPLE)
    find_package(PkgConfig REQUIRED)

    pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)

    pkg_check_modules(WEBKIT2GTK QUIET IMPORTED_TARGET webkit2gtk-4.1)

    if (NOT WEBKIT2GTK_FOUND)
        pkg_check_modules(WEBKIT2GTK QUIET IMPORTED_TARGET webkit2gtk-4.0)
    endif()
	
	if (UNIX AND NOT APPLE)
		find_package(CURL REQUIRED)
	endif()
endif()

set(PLUGIN_FORMATS VST3)

if(APPLE)
    list(APPEND PLUGIN_FORMATS AU)
endif()

juce_add_plugin(JPClipper
    COMPANY_NAME "Juicy Pads"
    COMPANY_WEBSITE "www.juicypads.com"
    COMPANY_EMAIL "info@juicypads.com"
    BUNDLE_ID "com.juicypads.jpclipper"

    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE

    COPY_PLUGIN_AFTER_BUILD FALSE

    # These 4-character IDs were not present in the .jucer file,
    # so they are chosen here for the CMake migration.
    # Keep them stable once you release the plugin.
    PLUGIN_MANUFACTURER_CODE Jcpy
    PLUGIN_CODE Jclp

    FORMATS ${PLUGIN_FORMATS}
    PRODUCT_NAME "JPClipper"
)

# Keeps compatibility with source files generated by Projucer that include <JuceHeader.h>.
juce_generate_juce_header(JPClipper)

target_sources(JPClipper
    PRIVATE
        Source/PluginProcessor.h
        Source/PluginProcessor.cpp
        Source/PluginEditor.h
        Source/PluginEditor.cpp
        Source/JPAssets.h
		Source/UI/UI.h
)

target_compile_definitions(JPClipper
    PUBLIC
        JUCE_STRICT_REFCOUNTEDPOINTER=1
        JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(JPClipper
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra

        $<$<AND:$<PLATFORM_ID:Linux>,$<TARGET_EXISTS:PkgConfig::GTK3>>:PkgConfig::GTK3>
        $<$<AND:$<PLATFORM_ID:Linux>,$<TARGET_EXISTS:PkgConfig::WEBKIT2GTK>>:PkgConfig::WEBKIT2GTK>
		$<$<PLATFORM_ID:Linux>:CURL::libcurl>

    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)
